import expense

def view_transactions():
    """READ: Displays all transactions and summarizes the net balance."""
    transactions = expense.load_data()
    if not transactions:
        print("\nNo transaction history found.")
        return

    print("\n=================== TRANSACTION HISTORY ===================")
    print(f"{'ID':<5} | {'Category':<15} | {'Type':<10} | {'Amount':<10} | {'Date':<12}")
    print("-" * 60)
    
    total_income = 0.0
    total_expense = 0.0
    
    for t in transactions:
        print(f"{t['id']:<5} | {t['category']:<15} | {t['type']:<10} | {t['amount']:<10.2f} | {t['date']:<12}")
        if t['type'].upper() == "INCOME":
            total_income += t['amount']
        elif t['type'].upper() == "EXPENSE":
            total_expense += t['amount']
            
    print("-" * 60)
    print(f"Total Income:  {total_income:.2f}")
    print(f"Total Expense: {total_expense:.2f}")
    print(f"Net Balance:   {(total_income - total_expense):.2f}")
    print("===========================================================")
